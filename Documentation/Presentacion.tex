% DIAPOSITIVA 10: MODELO PREDICTIVO OLS (TEORÍA)
% ---------------------------------------------------------
\begin{frame}{Modelo Predictivo OLS (Regresión Lineal Simple)}
    \vfill
    \begin{darkcard}[title=\color{cyanAccent}{Formulación Matemática MCO}]
        \fontsize{11pt}{15pt}\selectfont \color{textWhite}
        Evaluación de la relación entre cola inicial ($X$) y tiempo de salida ($Y$):
        \begin{equation*}
            \Large Y_i = a + b \cdot X_i + e_i
        \end{equation*}
        \textbf{Estimadores de Mínimos Cuadrados Ordinarios:}
        \begin{equation*}
            b = \frac{\sum (X_i - \bar{X})(Y_i - \bar{Y})}{\sum (X_i - \bar{X})^2}, \quad a = \bar{Y} - b \bar{X}
        \end{equation*}
    \end{darkcard}
    \vfill
\end{frame}

        \fontsize{11pt}{15pt}\selectfont \color{textWhite}
        \begin{equation*}
            \displaystyle \widehat{Y} = 14.1068 - 2.5395 \cdot X
        \end{equation*}
        \begin{itemize}
            \item \textbf{Intercepto ($a = 14.11$ min):} Espera estimada sin cola inicial ($X=0$).
            \item \textbf{Pendiente ($b = -2.54$ min/pax):} Cada alumno presente al llegar reduce la espera en $2.54$ minutos.
            \item \textbf{Significancia Estadística:} $p = 5.83 \times 10^{-43} < 0.001$, $F = 201.8$.
        \end{itemize}
    \end{darkcard}

\begin{frame}{Aprendizaje por Refuerzo: Algoritmo UCB1}
    \vfill
    \begin{darkcard}[title=\color{cyanAccent}{Formulación Matemática del Agente UCB1}]
        \fontsize{11pt}{15pt}\selectfont \color{textWhite}
        El agente selecciona la regla $A_t$ mediante la Cota Superior de Confianza:
        \begin{equation*}
            \displaystyle A_t = \arg\max_{a} \left[ \bar{Q}_t(a) + c \sqrt{\frac{\ln t}{N_t(a)}} \right]
        \end{equation*}
        \begin{itemize}
            \item \textbf{Explotación ($\bar{Q}_t(a)$):} Recompensa promedio histórica del brazo $a$.
            \item \textbf{Exploración ($c \sqrt{\frac{\ln t}{N_t(a)}}$):} Bonus de incertidumbre para probar reglas menos frecuentadas ($c = \sqrt{2}$).
        \end{itemize}
    \end{darkcard}
    \vfill
\end{frame}

% DIAPOSITIVA 18: FUNCIÓN DE RECOMPENSA NORMALIZADA
% ---------------------------------------------------------
\begin{frame}{Función de Recompensa Normalizada}
    \vfill
    \begin{darkcard}[title=\color{cyanAccent}{Penalización Multiobjetivo y Normalización Min-Max}]
        \fontsize{10.5pt}{14.5pt}\selectfont \color{textWhite}
        \begin{equation*}
            P_t = 0.4 \cdot \text{espera} + 0.4 \cdot \text{dejados} + 0.2 \cdot \text{vacíos}
        \end{equation*}
        Escalamiento invertido en el rango $R_t \in [0, 1]$:
        \begin{equation*}
            \displaystyle R_t = 1 - \frac{P_t - P_{\text{min}}}{P_{\text{max}} - P_{\text{min}}}
        \end{equation*}
        \begin{itemize}
            \item \textbf{$R_t \to 1$:} Despacho óptimo (mínima espera y alta ocupación).
            \item \textbf{$R_t \to 0$:} Despacho ineficiente.
        \end{itemize}
    \end{darkcard}
    \vfill
\end{frame}
