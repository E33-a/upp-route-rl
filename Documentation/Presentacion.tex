% DIAPOSITIVA 8: MODELO PREDICTIVO OLS (TEORÍA)
% ---------------------------------------------------------
\begin{frame}{Modelo Predictivo OLS (Regresión Lineal Simple)}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{darkcard}[title=\color{cyanAccent}{Formulación del Modelo MCO}]
                Relación entre alumnos esperando al llegar la unidad ($X$) y tiempo restante para salir ($Y$):
                \begin{equation*}
                    Y_i = a + b \cdot X_i + e_i
                \end{equation*}
                \fontsize{9pt}{12pt}\selectfont
                \textbf{Estimación de Coeficientes OLS:}
                \begin{equation*}
                    b = \frac{\sum (X_i - \bar{X})(Y_i - \bar{Y})}{\sum (X_i - \bar{X})^2}, \quad a = \bar{Y} - b \bar{X}
                \end{equation*}
            \end{darkcard}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{purplecard}{Propósito del Modelo}
                \fontsize{9.5pt}{12.5pt}\selectfont
                Proporcionar una \textbf{predicción del tiempo de espera esperado} a partir de la cola inicial observada al llegar la unidad, sirviendo como insumo al sistema de decisión.
            \end{purplecard}
        \end{column}
    \end{columns}
\end{frame}

% ---------------------------------------------------------
% DIAPOSITIVA 9: RESULTADOS Y AJUSTE DE REGRESIÓN OLS
% ---------------------------------------------------------
\begin{frame}{Resultados de la Regresión Lineal OLS}
    \fontsize{10pt}{13pt}\selectfont
    \begin{darkcard}[title=\color{cyanAccent}{Ecuación Estimada del Tiempo de Espera}]
        \begin{equation*}
            \widehat{Y} = 14.1068 - 2.5395 \cdot X
        \end{equation*}
        \begin{itemize}
            \item \textbf{Intercepto ($a = 14.11$ min):} Tiempo estimado si la combi llega sin alumnos en la cola inicial ($X=0$).
            \item \textbf{Pendiente ($b = -2.54$ min/alumno):} Por cada alumno en cola al llegar, el tiempo de espera restante se reduce en $2.54$ minutos.
            \item \textbf{Bondad de Ajuste ($R^2 = 0.121$):} El modelo explica el $12.1\%$ de la variabilidad, con alta significancia global ($F = 201.8, p = 5.83 \times 10^{-43} < 0.001$).
        \end{itemize}
    \end{darkcard}
\end{frame}

% DIAPOSITIVA 13: MARCO TEÓRICO UCB1
% ---------------------------------------------------------
\begin{frame}{Aprendizaje por Refuerzo: Algoritmo UCB1}
    \fontsize{10pt}{13pt}\selectfont
    \begin{darkcard}[title=\color{cyanAccent}{Formulación del Algoritmo UCB1 (Upper Confidence Bound)}]
        El agente selecciona la política $A_t$ maximizando la cota superior de confianza:
        \begin{equation*}
            A_t = \arg\max_{a} \left[ \bar{Q}_t(a) + c \sqrt{\frac{\ln t}{N_t(a)}} \right]
        \end{equation*}
        \begin{itemize}
            \item \textbf{Término de Explotación ($\bar{Q}_t(a)$):} Recompensa promedio obtenida históricamente por la regla $a$.
            \item \textbf{Término de Exploración ($c \sqrt{\frac{\ln t}{N_t(a)}}$):} Bonus de incertidumbre para brazos poco probados ($c = \sqrt{2}$).
        \end{itemize}
    \end{darkcard}
\end{frame}

% DIAPOSITIVA 15: FUNCIÓN DE RECOMPENSA NORMALIZADA
% ---------------------------------------------------------
\begin{frame}{Función de Recompensa Normalizada}
    \fontsize{9.5pt}{12.5pt}\selectfont
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{darkcard}[title=\color{cyanAccent}{Penalización Multiobjetivo}]
                \begin{equation*}
                    P_t = w_1 \cdot \text{espera} + w_2 \cdot \text{dejados} + w_3 \cdot \text{vacíos}
                \end{equation*}
                \textbf{Pesos del Servicio:}
                \begin{itemize}
                    \item $w_1 = 0.4$: Reducción del tiempo de espera.
                    \item $w_2 = 0.4$: Evitar alumnos varados.
                    \item $w_3 = 0.2$: Eficiencia de asientos.
                \end{itemize}
            \end{darkcard}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{purplecard}{Escalamiento Min-Max Invertido}
                Para estabilizar UCB1, la recompensa se normaliza en $R_t \in [0, 1]$:
                \begin{equation*}
                    R_t = 1 - \frac{P_t - P_{\text{min}}}{P_{\text{max}} - P_{\text{min}}}
                \end{equation*}
                \begin{itemize}
                    \item $R_t \to 1$: Despacho óptimo.
                    \item $R_t \to 0$: Despacho ineficiente.
                \end{itemize}
            \end{purplecard}
        \end{column}
    \end{columns}
\end{frame}
